# demo

A small planted repository.

Only `demo finish` needs Warden; every other command runs without it.

Without Warden, `demo finish` exits 2 and every other command runs as usual.

"""The command line."""

import sys

import cns


def run():
    if sys.argv[1:2] == ["finish"]:
        try:
            from .finisher import finish
        except ImportError:
            return 2
        return finish()
    return 0 if cns else 1
